💾 Relational Database Schema Design (PB-04)

The relational schema is structured with third normal form (3NF) compliance, supporting customer accounts, restaurant profiles, menu categories, dynamic pricing, and transactional order histories.

-- Database Schema DDL (MySQL 8.0 / Workbench) CREATE TABLE users ( id VARCHAR(50) PRIMARY KEY, name VARCHAR(200) NOT NULL, email VARCHAR(255) UNIQUE NOT NULL, password VARCHAR(255) NOT NULL, phone VARCHAR(30), role ENUM('Customer', 'Restaurant Admin', 'Delivery Agent', 'Super Admin') DEFAULT 'Customer', status ENUM('active', 'suspended') DEFAULT 'active', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE restaurants ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(200) NOT NULL, image VARCHAR(500), cuisine VARCHAR(100) NOT NULL, rating DECIMAL(3,1) DEFAULT 4.5, delivery_time VARCHAR(50) DEFAULT '25–35 mins', delivery_fee DECIMAL(8,2) DEFAULT 30.00, status ENUM('active', 'inactive') DEFAULT 'active' ); CREATE TABLE menu_items ( id INT AUTO_INCREMENT PRIMARY KEY, restaurant_id INT NOT NULL, name VARCHAR(200) NOT NULL, category VARCHAR(100) NOT NULL, description TEXT, price DECIMAL(10,2) NOT NULL, image VARCHAR(500), is_veg BOOLEAN DEFAULT FALSE, available BOOLEAN DEFAULT TRUE, FOREIGN KEY (restaurant_id) REFERENCES restaurants(id) ON DELETE CASCADE ); CREATE TABLE orders ( id VARCHAR(50) PRIMARY KEY, customer_name VARCHAR(200) NOT NULL, email VARCHAR(255) NOT NULL, restaurant_name VARCHAR(200) NOT NULL, subtotal DECIMAL(10,2) NOT NULL, delivery_fee DECIMAL(10,2) NOT NULL, platform_fee DECIMAL(10,2) DEFAULT 5.00, discount DECIMAL(10,2) DEFAULT 0.00, total DECIMAL(10,2) NOT NULL, status ENUM('pending', 'preparing', 'on-the-way', 'delivered', 'cancelled') DEFAULT 'pending', cancel_reason TEXT, cancelled_by ENUM('Customer', 'Admin', 'Restaurant', 'System'), refund_status ENUM('none', 'refunded', 'pending', 'not_applicable') DEFAULT 'none', refund_amount DECIMAL(10,2) DEFAULT 0.00, refund_ref VARCHAR(100), payment_method VARCHAR(50) NOT NULL, payment_status ENUM('success', 'pending', 'failed', 'refunded') DEFAULT 'success', delivery_address TEXT NOT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP );
